الدرس ٤: الجمع أو الطرح بتكوين عشرات
التطبيق
الإرشادات: تعاون مع زميلك لإيجاد جميع الطرق للحصول على ١٠.

| + ١ | = ١٠ |
| + ٢ | = ١٠ |
| + ٣ | = ١٠ |
| + ٤ | = ١٠ |
| + ٥ | = ١٠ |

| + ٦ | = ١٠ |
| + ٧ | = ١٠ |
| + ٨ | = ١٠ |
| + ٩ | = ١٠ |
| + ١٠ | = ١٠ |

أمثلة:
٧ + ٨ = ٦ + ٨ + ٤ = ٤ + ٦ + ١٠ = ١٤
٦ + ٨ = ١٤ ، إذاً ١٤ = ٨ + ٦

٩ + ٣ = ٩ + ١ = ١٠ ، ٢ + ١٠ = ١٢
إذاً، ٩ + ٣ = ١٢

١٥ - ٧ = ٧ - ٥ = ٢ ، ١٠ - ٢ = ٨
إذاً، ١٥ - ٧ = ٨
١٤ - ٩ = ١٤ - ٩ = ٥
إذاً، ١٤ - ٦ = ...........
