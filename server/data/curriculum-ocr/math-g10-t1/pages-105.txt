١ - الدوال المثلثية لأي زاويتين قياسيهما θ، (١٨٠° - θ)
من الشكل المقابل ب' (س'، ص') صورة النقطة ب (س، ص) بالانعكاس حول محور الصادات فيكون س' = -س، ص' = ص
لذلك فإن:
جا (١٨٠° - θ) = جا θ، قتا (١٨٠° - θ) = قتا θ
جتا (١٨٠° - θ) = - جتا θ، قا (١٨٠° - θ) = - قا θ
ظا (١٨٠° - θ) = - ظا θ، ظتا (١٨٠° - θ) = - ظتا θ

فمثلاً: جتا ١٢٠° = جتا (١٨٠° - ٦٠°) = - جتا ٦٠° = - ١/٢
جا ١٣٥° = جا (١٨٠° - ٤٥°) = جا ٤٥° = ١/√٢

١) أوجد ظا ١٣٥°، جا ١٢٠°، جتا ١٥٠°
لاحظ أن: (١٨٠° - θ) + θ = ١٨٠°
