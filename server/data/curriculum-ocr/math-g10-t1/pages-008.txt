إذا كان أ، ب عددين حقيقيين فإن العدد ع حيث ع = أ + ب ت يسمى عدداً مركباً، وتسمى أ بالجزء الحقيقي للعدد المركب ع، ب ت بالجزء التخيلي للعدد المركب ع.
وإذا كانت ب = ٠ فإن العدد ع = أ يكون حقيقياً، وإذا كانت أ = ٠ فإن العدد ع = ب ت يكون تخيلياً
حيث ب ≠ صفر.

مثال
(٢) حل المعادلة ٩س٢ + ١٢٥ = ٦١
الحل
المعادلة ٩س٢ + ١٢٥ = ٦١
٩س٢ + ١٢٥ - ١٢٥ = ٦١ - ١٢٥ بإضافة (- ١٢٥) إلى طرفي المعادلة
٩س٢ = - ٦٤ بقسمة طرفي المعادلة على ٩
س٢ = - ٦٤ / ٩
س = ± جذر (- ٦٤ / ٩) بأخذ الجذر التربيعي
س = ± ٨ / ٣ ت تعريف العدد المركب

حاول أن تحل
(٢) حل كلاً من المعادلات الآتية:
أ) ٣س٢ + ٢٧ = ٠
ب) ٥س٢ + ٢٤٥ = ٠
ج) ٤س٢ + ١٠٠ = ٧٥
تساوى عددين مركبين Equality of two complex numbers
يتساوى العددان المركبان إذا وفقط إذا تساوى الجزآن الحقيقيان وتساوى الجزآن التخيليان.
إذا كان: أ + ب ت = جـ + د ت فإن: أ = جـ ، ب = د والعكس صحيح
