رابعًا: إذا كانت θ = ٢٧٠° = ٣π/٢
جتا ٢٧٠° = ٠ ، جا ٢٧٠° = -١ ، فإن: ب(٠، -١)
ظا ٢٧٠° = -١/صفر (غير معرف)

حاول أن تحل
٤) في الأشكال التالية حدد إحداثيي النقطة ب لكل شكل واستنتج الدوال المثلثية لقياسات الزوايا ٣٠°، ٦٠°، ٤٥°

مثال
٥) أثبت بدون استخدام الآلة الحاسبة أن: جا ٦٠° جتا ٣٠° - جتا ٦٠° جا ٣٠° = جا π/٦
الحل
تعلم أن جا ٣٠° = ١/٢ ، جتا ٣٠° = √٣/٢ ، جا ٦٠° = √٣/٢ ، جتا ٦٠° = ١/٢
.. الطرف الأيمن = √٣/٢ × √٣/٢ - ١/٢ × ١/٢ = ٣/٤ - ١/٤ = ٢/٤ = ١/٢ (١)
.. ٤٥° = π/٤ ، جا ٤٥° = ١/√٢
الطرف الأيسر = جا π/٦ = (١/٢)^٢ = ١/٢ (٢)
من (١)، (٢) .. الطرفان متساويان.

حاول أن تحل
٥) أوجد قيمة: ٣ جا ٣٠° جا ٢ ٦٠° - جتا ٠° قا ٦٠° + جا ٢٧٠° جتا ٢ ٤٥°
تفكير ناقد: إذا كانت الزاوية التي قياسها θ مرسومة في الوضع القياسي، وكان جتا θ = ١/٢، جا θ = √٣/٢
هل من الممكن أن يكون θ = ٢٤٠°؟ وضح ذلك.

أ) ١ - ٢ جا ٢ ٩٠° = جتا ١٨٠°
