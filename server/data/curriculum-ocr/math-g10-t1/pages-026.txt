مثال
٥ مثل بيانيًا د حيث د(س) = ٤ س٢ - ٤ س + ١ ، ثم عين إشارة الدالة د.
الحل
المميز (ب٢ - ٤ أ جـ) = (-٤)٢ - ٤ × ٤ × ١
= ١٦ - ١٦ = ٠
لذلك فإن المعادلة ٤ س٢ - ٤ س + ١ = ٠ لها جذران متساويان.
بالتحليل : (٢ س - ١)٢ = ٠
بوضع : ٢ س - ١ = ٠ تكون س = ١/٢
د(س) > ٠ عندما س ≠ ١/٢ ، د(س) = ٠ عندما س = ١/٢
حاول أن تحل
٥ مثل بيانيًا د، حيث د(س) = ٤ س٢ - ١٢ س - ٩ ثم عين إشارة الدالة د.

مثال
٦ اثبت أنه لجميع قيم س ∈ ع يكون جذرا المعادلة ٢ س٢ - ك س + ك - ٣ = صفر حقيقيين مختلفين
الحل
المميز (ب٢ - ٤ أ جـ) = (-ك)٢ - ٤ × ٢ × (ك - ٣) = ك٢ - ٨ ك + ٢٤
يكون جذرا المعادلة حقيقيين مختلفين إذا كان المميز موجبًا
نبحث إشارة المقدار ص = ك٢ - ٨ ك + ٢٤
فيكون مميز المعادلة ك٢ - ٨ ك + ٢٤ = ٠ هو:
(٨-)٢ - ٤ × ١ × ٢٤ = ٦٤ - ٩٦ = -٣٢ < ٠
لذلك فإن المعادلة ك٢ - ٨ ك + ٢٤ = ٠ ليس لها جذور حقيقية
∴ إشارة المقدار ص = ك٢ - ٨ ك + ٢٤ موجبة لكل س ∈ ع (لماذا)؟
فيكون مميز المعادلة ٢ س٢ - ك س + ك - ٣ = صفر موجب لكل س ∈ ع
∴ جذرا المعادلة ٢ س٢ - ك س + ك - ٣ = ٠ حقيقيان مختلفان لكل س ∈ ع

تحقق من فهمك
١ عين إشارة كل دالة من الدوال الآتية:
أ) د(س) = ٢ س - ٣
ب) د(س) = ٤ - س
جـ) د(س) = س٢ - ٤
د) د(س) = ١ - س٢
هـ) د(س) = ٤ + ٤ س + س٢
و) د(س) = ٣ - ٢ س٢ + ٤
